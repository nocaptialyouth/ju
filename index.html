📜 2단계: 구글 스프레드시트 양방향 자동화 매크로 (Apps Script V3.0)

이 코드는 웹에서 내역 기입 시 ① 기존 적금 시트 금액 자동 반영, ② 매일가계부 통합장부 시트 자동 추가, ③ 새로운 시트(탭) 자동 생성을 모두 지원합니다.

Code.gs (구글 시트 Apps Script 편집기에 덮어쓰기) 
/**
 * 📒 매일가계부 & 주원 가족 자산 통합 Google Apps Script (V3.0 양방향 자동 반영 엔진)
 * - 웹 대시보드에서 내역 추가/수정 시 스프레드시트 금액 자동 기입 및 새 시트 자동 생성 지원
 */

function onOpen(e) {
  try {
    var ui = SpreadsheetApp.getUi();
    ui.createMenu('📒 매일가계부 관리')
      .addItem('⚡ 자산합계 새로고침', 'fillAllAccounts')
      .addToUi();
  } catch (err) {}
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0];
  var data = sheet.getDataRange().getDisplayValues();
  
  var memoSheet = ss.getSheetByName('일정메모');
  var memoData = memoSheet ? memoSheet.getDataRange().getDisplayValues() : [];

  var webLedgerSheet = ss.getSheetByName('매일가계부_웹입력');
  var webLedgerData = webLedgerSheet ? webLedgerSheet.getDataRange().getDisplayValues() : [];

  return ContentService.createTextOutput(JSON.stringify({
    grid: data,
    memos: memoData,
    webLedger: webLedgerData
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var params = JSON.parse(e.postData.contents);
    var nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");

    // 1. 새로운 시트(탭) 생성 요청 처리
    if (params.action === 'create_new_sheet' || params.action === 'create_edit_history_sheet') {
      var targetName = params.sheetName || ('가계부시트_' + Utilities.formatDate(new Date(), "Asia/Seoul", "yyyyMMdd_HHmm"));
      var newSheet = ss.getSheetByName(targetName);
      if (!newSheet) {
        newSheet = ss.insertSheet(targetName);
      } else {
        newSheet.clear();
      }
      var headers = params.headers || ['NO', '기록내역(상품명)', '결제일/입금일', '금액', '소분류', '대분류', '상태', '명의자/출금계좌', '거래수단/비고', '동기화일시'];
      newSheet.appendRow(headers);
      newSheet.getRange(1, 1, 1, headers.length).setBackground('#fbf3db').setFontWeight('bold');

      var rows = params.records || [];
      for (var i = 0; i < rows.length; i++) {
        var r = rows[i];
        newSheet.appendRow([
          i + 1,
          r.title || r.productName || '',
          r.date || '',
          Number(r.amount) || 0,
          r.subCategory || r.month || '',
          r.mainCategory || '저축/투자',
          r.status || '입금완료',
          r.owner || '가족공용',
          r.method || r.memo || '',
          nowStr
        ]);
      }
      if (rows.length > 0) {
        newSheet.appendRow(['', '합계', '', '=SUM(D2:D' + (rows.length + 1) + ')', '', '', '', '', '', '']);
      }
      return jsonResponse({ status: 'success', sheetName: targetName });
    }

    // 2. 적금 입금 / 내역 수정 시 메인 시트 해당 월 칸 자동 업데이트 + 웹입력 장부 기록
    if (params.action === 'deposit' || params.action === 'update_record' || params.action === 'add_ledger_entry') {
      var rec = params.record || params;
      var prodName = rec.productName || rec.title || '';
      var monthStr = rec.month || '';
      var dateStr = rec.date || '';
      var amountNum = Number(rec.amount) || 0;

      // (A) 메인 적금 시트(첫번째 탭)에서 상품명과 월을 찾아 날짜/금액 자동 기입
      var mainSheet = ss.getSheets()[0];
      var grid = mainSheet.getDataRange().getDisplayValues();
      var updatedInMain = false;

      if (prodName && monthStr) {
        for (var r = 0; r < grid.length - 1; r++) {
          for (var c = 0; c < grid[r].length; c++) {
            var cellText = (grid[r][c] || '').toString().trim();
            if (cellText && (cellText.indexOf(prodName) !== -1 || prodName.indexOf(cellText) !== -1)) {
              // 주변 열에서 '월' 헤더 탐색
              var subRow = grid[r + 1] || [];
              for (var sc = Math.max(0, c - 2); sc <= Math.min(subRow.length - 1, c + 3); sc++) {
                if ((subRow[sc] || '').toString().trim() === '월') {
                  for (var mr = r + 2; mr < grid.length; mr++) {
                    var mVal = (grid[mr][sc] || '').toString().trim();
                    if (mVal === '총액' || mVal === '월') break;
                    if (mVal === monthStr || mVal.indexOf(monthStr) !== -1) {
                      mainSheet.getRange(mr + 1, sc + 2).setValue(rec.status === 'unpaid' ? '미입금' : dateStr);
                      if (amountNum > 0) mainSheet.getRange(mr + 1, sc + 3).setValue(amountNum);
                      updatedInMain = true;
                      break;
                    }
                  }
                }
              }
            }
          }
        }
      }

      // (B) [매일가계부_웹입력] 시트에 모든 추가/수정 내역 누적 기록 (시트가 없으면 자동 생성)
      var ledgerSheet = ss.getSheetByName('매일가계부_웹입력');
      if (!ledgerSheet) {
        ledgerSheet = ss.insertSheet('매일가계부_웹입력');
        ledgerSheet.appendRow(['기록내역(상품명)', '월/회차', '결제/입금일자', '금액', '소분류', '대분류', '상태', '명의자/계좌', '거래수단/비고', '등록일시']);
        ledgerSheet.getRange(1, 1, 1, 10).setBackground('#fbf3db').setFontWeight('bold');
      }
      ledgerSheet.appendRow([
        prodName,
        monthStr,
        dateStr,
        amountNum,
        rec.subCategory || '예적금/가계부',
        rec.mainCategory || '저축/지출',
        rec.status || 'deposited',
        rec.owner || '가족공용',
        rec.method || rec.memo || '웹 대시보드 기입',
        nowStr
      ]);

      return jsonResponse({ status: 'success', updatedInMain: updatedInMain });
    }

    // 3. 신규 적금 상품 등록 시 [적금리스트] 탭에 자동 행 추가
    if (params.action === 'new_product') {
      var listSheet = ss.getSheetByName('적금리스트') || ss.getSheets()[1];
      if (listSheet) {
        var isJuwon = params.owner === '주원' ? 'O' : '';
        var isJihyeon = params.owner === '지헌' ? 'O' : '';
        var isJunyeong = params.owner === '준영' ? 'O' : '';
        listSheet.appendRow(['', '', params.name, isJuwon, isJihyeon, isJunyeong, Number(params.monthly) || 0, Number(params.initial) || 0, params.startDate || '-', params.endDate || '-']);
      }
      return jsonResponse({ status: 'success' });
    }

    // 4. 병원비/조리원비 지출 추가 시 해당 시트에 자동 행 추가
    if (params.action === 'add_hospital_expense') {
      var hospSheet = ss.getSheetByName('조리원/병원비') || ss.getSheets()[2];
      if (hospSheet) {
        hospSheet.appendRow([params.date, params.category, params.details, Number(params.amount) || 0, params.note || '-']);
      }
      return jsonResponse({ status: 'success' });
    }

    // 5. 일정/메모 저장
    var memoSheet = ss.getSheetByName('일정메모');
    if (!memoSheet) {
      memoSheet = ss.insertSheet('일정메모');
      memoSheet.appendRow(['날짜', '시간', '카테고리', '내용', '등록일시']);
    }
    memoSheet.appendRow([params.date, params.time, params.category, params.content, nowStr]);
    return jsonResponse({ status: 'success' });

  } catch (err) {
    return jsonResponse({ status: 'error', message: err.message });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function fillAllAccounts() {
  SpreadsheetApp.getUi().alert('자산 합계 동기화가 완료되었습니다!');
}
