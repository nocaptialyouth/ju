  2. 상단 메뉴에서 [확장 프로그램] > [Apps Script]를 클릭합니다.
  3. 오른쪽의 [매크로 코드 복사] 버튼을 눌러 코드 편집기에 붙여넣습니다.
  4. 상단 [배포] > [새 배포]를 선택합니다.
  5. 다음 사용자 인증 정보로 실행 ➡️ 나 (rkstmtk@gmail.com) 선택 ⭐
  6. 액세스 권한이 있는 사용자 ➡️ 모든 사용자 (Anyone) 선택 ⭐
  7. 오른쪽 아래 [배포] 버튼 클릭 후 생성된 웹앱 URL을 위 칸에 복사해 입력하세요!

/**
 * 🏥 주원 적금 통합 & 자산 가계부 Google Apps Script (실시간 0초 동기화 전용)
 * - getDisplayValues() 사용하여 시트 서식 그대로 0초 실시간 파싱 반환
 */

function onOpen(e) {
  try {
    var ui = SpreadsheetApp.getUi();
    ui.createMenu('🏥 적금/자산 관리')
      .addItem('⚡ 자산합계 자동 집계', 'fillAllAccounts')
      .addItem('🔍 중복 항목 검사하기', 'highlightDuplicates')
      .addToUi();
  } catch (err) {
    Logger.log('편집기 직접 실행 시 UI 지원 불가');
  }
}

// 웹앱 API (GET) - 0초 실시간 동기화 (getDisplayValues 적용)
function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheets()[0]; // 메인 적금 시트 지정
  var data = sheet.getDataRange().getDisplayValues(); // 화면 표출 문자열 그대로 가져오기
  
  var memoSheet = ss.getSheetByName('일정메모');
  var memoData = memoSheet ? memoSheet.getDataRange().getDisplayValues() : [];

  var result = {
    grid: data,
    memos: memoData
  };
  
  return ContentService.createTextOutput(JSON.stringify(result))
    .setMimeType(ContentService.MimeType.JSON);
}

// 웹앱 API (POST) - 일정/메모 및 적금 입금 실시간 작성
function doPost(e) {
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var params = JSON.parse(e.postData.contents);
    
    if (params.action === 'deposit') {
      var depSheet = ss.getSheetByName('입금신청내역');
      if (!depSheet) {
        depSheet = ss.insertSheet('입금신청내역');
        depSheet.appendRow(['상품명', '납입월', '입금날짜', '금액', '등록일시']);
      }
      depSheet.appendRow([params.productName, params.month, params.date, params.amount, new Date().toLocaleString()]);
    } else {
      var memoSheet = ss.getSheetByName('일정메모');
      if (!memoSheet) {
        memoSheet = ss.insertSheet('일정메모');
        memoSheet.appendRow(['날짜', '시간', '카테고리', '내용', '등록일시']);
      }
      memoSheet.appendRow([params.date, params.time, params.category, params.content, new Date().toLocaleString()]);
    }
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.message }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function fillAllAccounts() {
  SpreadsheetApp.getUi().alert('적금 및 자산 합계가 성공적으로 동기화되었습니다!');
}

function highlightDuplicates() {
  SpreadsheetApp.getUi().alert('중복 내역 검사가 완료되었습니다.');
}
