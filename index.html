/**
 * 📒 매일가계부 & 주원 가족 자산 통합 Google Apps Script (V5.0 — 단방향 자동저장 & 상품/내역 삭제 지원 완결판)
 * - 스프레드시트 고유 ID: 16xlN-rZwRRsOhdylZGZZL1-FF3flTthARoE6Tr4Xwrg (량신)
 * - 웹(ju-mu-five.vercel.app)에서 기입·수정·삭제한 모든 내역이 구글 시트에 100% 자동 저장됩니다.
 * - 적금 상품 삭제(delete_product), 가계부 내역 삭제(delete_record), 병원비 삭제 완벽 대응.
 */

var SPREADSHEET_ID = '16xlN-rZwRRsOhdylZGZZL1-FF3flTthARoE6Tr4Xwrg';

/**
 * ✅ 테스트 실행 함수 (Apps Script 편집기에서 [실행] 클릭 시 오류 없이 즉시 점검)
 */
function testConnection() {
  var ss = getSS();
  if (!ss) {
    Logger.log("❌ 스프레드시트를 열 수 없습니다. 시트 ID를 확인해주세요.");
    return "FAIL";
  }
  Logger.log("✅ 스프레드시트 정상 연결 완료: [" + ss.getName() + "]");
  var gridSheet = getSheetByGid(ss, 661362148, 1);
  var masterSheet = getSheetByGid(ss, 7284588, 0);
  Logger.log("  • 월별 시트: " + (gridSheet ? gridSheet.getName() : "없음"));
  Logger.log("  • 적금리스트: " + (masterSheet ? masterSheet.getName() : "없음"));
  return "SUCCESS";
}

function getSS() {
  try {
    var active = SpreadsheetApp.getActiveSpreadsheet();
    if (active) return active;
  } catch(e) {}
  try {
    return SpreadsheetApp.openById(SPREADSHEET_ID);
  } catch(e2) {
    return null;
  }
}

function getSheetByGid(ss, gid, fallbackIdx) {
  if (!ss) ss = getSS();
  if (!ss) return null;
  try {
    var sheets = ss.getSheets();
    if (!sheets || sheets.length === 0) return null;
    for (var i = 0; i < sheets.length; i++) {
      if (sheets[i].getSheetId() == gid) return sheets[i];
    }
    return sheets[fallbackIdx || 0] || sheets[0];
  } catch (err) {
    return null;
  }
}

function cleanProdKey(str) {
  if (!str) return '';
  return str.toString().split('(')[0].replace(/\s+/g, '').trim();
}

function doGet(e) {
  var ss = getSS();
  var gridSheet = getSheetByGid(ss, 661362148, 1);
  var masterSheet = getSheetByGid(ss, 7284588, 0);
  var memoSheet = ss ? ss.getSheetByName('일정메모') : null;
  var webLedgerSheet = ss ? ss.getSheetByName('매일가계부_웹입력') : null;

  return ContentService.createTextOutput(JSON.stringify({
    version: "5.0",
    status: "ok",
    sheetName: ss ? ss.getName() : "미연결",
    grid: gridSheet ? gridSheet.getDataRange().getDisplayValues() : [],
    master: masterSheet ? masterSheet.getDataRange().getDisplayValues() : [],
    memos: memoSheet ? memoSheet.getDataRange().getDisplayValues() : [],
    webLedger: webLedgerSheet ? webLedgerSheet.getDataRange().getDisplayValues() : []
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var ss = getSS();
    if (!ss) {
      return jsonResponse({ status: 'error', message: '스프레드시트를 열 수 없습니다.' });
    }
    var params = {};
    if (e && e.postData && e.postData.contents) {
      try {
        params = JSON.parse(e.postData.contents);
      } catch(pErr) {
        params = e.parameter || {};
      }
    } else if (e && e.parameter) {
      params = e.parameter;
    }

    var act = params.subAction || params.action || 'deposit';
    var nowStr = Utilities.formatDate(new Date(), "Asia/Seoul", "yyyy-MM-dd HH:mm:ss");

    // 1. [신규 기능] 적금 상품 삭제 요청 (delete_product)
    if (act === 'delete_product') {
      var prodToDelete = cleanProdKey(params.productName || params.name || '');
      var masterSheet = getSheetByGid(ss, 7284588, 0);
      var deletedCount = 0;
      if (masterSheet && prodToDelete) {
        var mGrid = masterSheet.getDataRange().getDisplayValues();
        for (var mr = mGrid.length - 1; mr >= 0; mr--) {
          var rowName = cleanProdKey(mGrid[mr][3] || mGrid[mr][2] || '');
          if (rowName && (rowName.indexOf(prodToDelete) !== -1 || prodToDelete.indexOf(rowName) !== -1)) {
            masterSheet.deleteRow(mr + 1);
            deletedCount++;
          }
        }
      }
      var ledgerSheet = ss.getSheetByName('매일가계부_웹입력');
      if (ledgerSheet) {
        ledgerSheet.appendRow(['[상품삭제] ' + (params.productName || params.name), '-', '-', 0, '적금상품삭제', '삭제', '삭제완료', '웹동기화', '관리자 삭제', nowStr]);
      }
      return jsonResponse({ status: 'success', version: '5.0', deleted: prodToDelete, deletedCount: deletedCount });
    }

    // 2. [신규 기능] 가계부 특정 내역 삭제 / 취소 (delete_record)
    if (act === 'delete_record') {
      var rec = params.record || params;
      var prodName = rec.productName || params.productName || '';
      var monthStr = (rec.month || params.month || '').toString().trim();
      var prodKey = cleanProdKey(prodName);

      var gridSheet = getSheetByGid(ss, 661362148, 1);
      if (gridSheet && prodKey && monthStr) {
        var grid = gridSheet.getDataRange().getDisplayValues();
        for (var r = 0; r < grid.length - 1; r++) {
          var subRow = grid[r + 1] || [];
          for (var sc = 0; sc < subRow.length; sc++) {
            if ((subRow[sc] || '').toString().trim() === '월') {
              var headerCandidate = ((grid[r][sc] || '') + ' ' + (grid[r][sc + 1] || '')).trim();
              var headerKey = cleanProdKey(headerCandidate);
              if (headerKey && (headerKey.indexOf(prodKey) !== -1 || prodKey.indexOf(headerKey) !== -1)) {
                for (var mr = r + 2; mr < grid.length; mr++) {
                  var mVal = (grid[mr][sc] || '').toString().trim();
                  if (mVal === '총액' || mVal === '총급여' || mVal === '월') break;
                  if (mVal === monthStr) {
                    gridSheet.getRange(mr + 1, sc + 2).setValue('');
                    gridSheet.getRange(mr + 1, sc + 3).setValue('');
                    break;
                  }
                }
              }
            }
          }
        }
      }
      return jsonResponse({ status: 'success', version: '5.0', message: '내역 삭제 완료' });
    }

    // 3. 병원비 항목 삭제 (delete_hospital_expense)
    if (act === 'delete_hospital_expense') {
      var hospSheet = getSheetByGid(ss, 1541600042, 2);
      if (hospSheet && params.details) {
        var hGrid = hospSheet.getDataRange().getDisplayValues();
        var targetDet = params.details.toString().trim();
        for (var hr = hGrid.length - 1; hr >= 0; hr--) {
          if ((hGrid[hr][3] || '').toString().trim() === targetDet) {
            hospSheet.deleteRow(hr + 1);
            break;
          }
        }
      }
      return jsonResponse({ status: 'success', version: '5.0' });
    }

    // 4. 새로운 시트(탭) 생성 요청 처리
    if (act === 'create_new_sheet' || act === 'create_edit_history_sheet') {
      var targetName = (params.sheetName || ('매일가계부_' + Utilities.formatDate(new Date(), "Asia/Seoul", "yyyyMMdd_HHmm"))).toString().trim();
      var newSheet = ss.getSheetByName(targetName);
      if (!newSheet) {
        newSheet = ss.insertSheet(targetName);
      } else {
        newSheet.clear();
      }
      var headers = ['NO', '기록내역(상품명)', '결제일/입금일', '금액', '소분류', '대분류', '상태', '명의자/출금계좌', '거래수단/비고', '동기화일시'];
      newSheet.appendRow(headers);
      newSheet.getRange(1, 1, 1, headers.length).setBackground('#fbf3db').setFontWeight('bold');

      var rows = params.records || [];
      for (var i = 0; i < rows.length; i++) {
        var r = rows[i];
        newSheet.appendRow([
          i + 1,
          r.title || r.productName || '',
          r.date || '',
          Number(r.amount) || 0,
          r.subCategory || r.month || '',
          r.mainCategory || '저축/투자',
          r.status || '입금완료',
          r.owner || '가족공용',
          r.method || r.memo || '',
          nowStr
        ]);
      }
      if (rows.length > 0) {
        newSheet.appendRow(['', '합계', '', '=SUM(D2:D' + (rows.length + 1) + ')', '', '', '', '', '', '']);
      }
      return jsonResponse({ status: 'success', version: '5.0', sheetName: targetName, rowCount: rows.length });
    }

    // 5. 적금 입금 / 내역 수정 / 웹 가계부 기입 처리
    if (act === 'deposit' || act === 'update_record' || act === 'add_ledger_entry') {
      var rec = params.record || params;
      var prodName = rec.productName || params.productName || rec.title || '';
      var monthStr = (rec.month || params.month || '').toString().trim();
      var dateStr = (rec.date || params.date || '').toString().trim();
      var amountNum = Number(rec.amount !== undefined ? rec.amount : params.amount) || 0;
      var statusStr = rec.status || params.status || 'deposited';
      var prodKey = cleanProdKey(prodName);

      // (A) 월별 적금 통합 리스트 시트(gid=661362148)에서 해당 상품 & 월 셀 찾아 직접 기입
      var gridSheet = getSheetByGid(ss, 661362148, 1);
      var grid = gridSheet ? gridSheet.getDataRange().getDisplayValues() : [];
      var updatedBlockSum = -1;

      if (gridSheet && prodKey && monthStr) {
        for (var r = 0; r < grid.length - 1; r++) {
          var subRow = grid[r + 1] || [];
          for (var sc = 0; sc < subRow.length; sc++) {
            if ((subRow[sc] || '').toString().trim() === '월') {
              var headerCandidate = ((grid[r][sc] || '') + ' ' + (grid[r][sc + 1] || '')).trim();
              var headerKey = cleanProdKey(headerCandidate);
              if (headerKey && (headerKey.indexOf(prodKey) !== -1 || prodKey.indexOf(headerKey) !== -1)) {
                var runningSum = 0;
                for (var mr = r + 2; mr < grid.length; mr++) {
                  var mVal = (grid[mr][sc] || '').toString().trim();
                  if (mVal === '총액' || mVal === '총급여' || mVal === '월') {
                    if (mVal === '총액' || mVal === '총급여') {
                      var totalCell = gridSheet.getRange(mr + 1, sc + 3);
                      if (!totalCell.getFormula()) totalCell.setValue(runningSum);
                      updatedBlockSum = runningSum;
                    }
                    break;
                  }
                  if (mVal === monthStr) {
                    if (statusStr === 'unpaid') {
                      gridSheet.getRange(mr + 1, sc + 2).setValue('');
                      gridSheet.getRange(mr + 1, sc + 3).setValue('');
                    } else {
                      gridSheet.getRange(mr + 1, sc + 2).setValue(dateStr);
                      gridSheet.getRange(mr + 1, sc + 3).setValue(amountNum);
                      runningSum += amountNum;
                    }
                  } else {
                    var existingDate = (grid[mr][sc + 1] || '').toString().trim();
                    var existingAmt = Number((grid[mr][sc + 2] || '').toString().replace(/[^0-9.-]+/g, '')) || 0;
                    if (existingDate && existingDate !== '미입금' && existingAmt > 0) {
                      runningSum += existingAmt;
                    }
                  }
                }
              }
            }
          }
        }
      }

      // (B) 적금리스트 시트(gid=7284588)의 총금액(I열)도 함께 갱신
      if (prodKey && updatedBlockSum >= 0) {
        var masterSheet = getSheetByGid(ss, 7284588, 0);
        if (masterSheet) {
          var mGrid = masterSheet.getDataRange().getDisplayValues();
          for (var mr2 = 0; mr2 < mGrid.length; mr2++) {
            var mName = cleanProdKey(mGrid[mr2][3] || mGrid[mr2][2] || '');
            if (mName && (mName.indexOf(prodKey) !== -1 || prodKey.indexOf(mName) !== -1)) {
              var mTotalCell = masterSheet.getRange(mr2 + 1, 9); // Col I (총금액)
              if (!mTotalCell.getFormula()) mTotalCell.setValue(updatedBlockSum);
              break;
            }
          }
        }
      }

      // (C) [매일가계부_웹입력] 시트에 모든 웹 입력 이력 누적 기록
      var ledgerSheet = ss.getSheetByName('매일가계부_웹입력');
      if (!ledgerSheet) {
        ledgerSheet = ss.insertSheet('매일가계부_웹입력');
        ledgerSheet.appendRow(['기록내역(상품명)', '월/회차', '결제/입금일자', '금액', '소분류', '대분류', '상태', '명의자/계좌', '거래수단/비고', '등록일시']);
        ledgerSheet.getRange(1, 1, 1, 10).setBackground('#fbf3db').setFontWeight('bold');
      }
      ledgerSheet.appendRow([
        prodName, monthStr, dateStr, amountNum,
        rec.subCategory || '예적금/가계부',
        rec.mainCategory || '저축/지출',
        statusStr,
        rec.owner || '가족공용',
        rec.method || rec.memo || '웹 대시보드 기입',
        nowStr
      ]);

      return jsonResponse({ status: 'success', version: '5.0', updatedBlockSum: updatedBlockSum });
    }

    // 6. 신규 적금 상품 등록 시 [적금리스트](gid=7284588)에 자동 행 추가
    if (act === 'new_product') {
      var listSheet = getSheetByGid(ss, 7284588, 0);
      if (listSheet) {
        var isJuwon = params.owner === '주원' ? 'O' : 'X';
        var isJihyeon = params.owner === '지헌' ? 'O' : 'X';
        var isJunyeong = params.owner === '준영' ? 'O' : 'X';
        listSheet.appendRow(['', '', '', params.name, isJuwon, isJihyeon, isJunyeong, Number(params.monthly) || 0, Number(params.initial) || 0, params.startDate || '-', params.endDate || '-']);
      }
      return jsonResponse({ status: 'success', version: '5.0' });
    }

    // 7. 병원비/조리원비 지출 추가 시 (gid=1541600042) 자동 행 추가
    if (act === 'add_hospital_expense') {
      var hospSheet = getSheetByGid(ss, 1541600042, 2);
      if (hospSheet) {
        hospSheet.appendRow(['', params.date, params.category, params.details, Number(params.amount) || 0, params.note || '-']);
      }
      return jsonResponse({ status: 'success', version: '5.0' });
    }

    // 8. 일정/메모 저장
    if (params.date && params.content) {
      var memoSheet = ss.getSheetByName('일정메모');
      if (!memoSheet) {
        memoSheet = ss.insertSheet('일정메모');
        memoSheet.appendRow(['날짜', '시간', '카테고리', '내용', '등록일시']);
      }
      memoSheet.appendRow([params.date, params.time || '', params.category || '메모', params.content, nowStr]);
    }
    return jsonResponse({ status: 'success', version: '5.0' });

  } catch (err) {
    return jsonResponse({ status: 'error', message: err.message });
  }
}

function jsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
